def escape_heading(text):
    return text.replace('&', r'\&').replace('_', r'\_').replace('%', r'\%').replace('—', '---').replace('–', '--').replace('’', "'")


def main():
    mapping = json.loads((R / 'RESPONSE_MAP.json').read_text())
    body = r'''\begin{frontmatter}
\title{Response to the Referee: Neural Bellman Operators}
\runtitle{NBO Response}
\end{frontmatter}
\section*{The revision and the status of this draft}
We retain Neural Bellman Operators as the paper's subject and address the report
through its NBO-centered path. The current manuscript combines the original
economic applications, one evaluation--improvement algorithm, the general
verification arguments, a finite Bellman mechanism, and direct computational
evidence. Complete models and proofs remain in the current supplement. The
unfavorable preceding comparisons remain visible and the exact numerical record
is preserved.

This is a working response. The finite-observation experiment is complete; the
new primary method comparison and numerical mechanism assessment are still
pending. The responses below distinguish implemented design and proved
statements from executed findings. They will be completed with the actual
source-bound outputs, including failures, before this draft is presented as a
completed numerical revision. The report is treated as the advisory review
stored in the repository, rather than as an Econometrica editorial decision.
'''
    for comment in mapping['comments']:
        status, reply = REPLIES[comment['id']]
        comment['current_status'] = status
        comment['current_response_tex'] = reply
        body += '\n\\section*{' + escape_heading(comment['id'] + ': ' + comment['review_heading']) + '}\n'
        body += reply + '\n'
    mapping['status'] = 'current editorial and theorem response; protected-observation evidence complete; primary and mechanism outcomes pending'
    (R / 'RESPONSE_MAP.json').write_text(json.dumps(mapping, indent=2) + '\n')
    (M / 'response_body.tex').write_text(body)


if __name__ == '__main__':
    main()
