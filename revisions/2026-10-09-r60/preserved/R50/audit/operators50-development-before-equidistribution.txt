"""Cost-directed terminal repair and residual-driven tensor FVI.

The gate compares the ACTUAL conditional terminal cost, not a Bellman
certificate. All accepted actions are feasible throughout the acquired cell.
This file never changes a frozen baseline policy or result.
"""

import service49 as s
np=s.np; I=s.I; BETA=s.BETA
QUANTUM=4096


def deterministic_next(x:I,a:I)->I:
    """Unclipped invariant transition without the common innovation."""
    d=x.lo.shape[-1];out=[]
    for j in range(d):
        u,v=s.col(x,j),s.col(x,(j+1)%d)
        out.append(F(1,16)+u/2+v/8+u*(1-v)/16+(F(1,2) if j%2==0 else F(1,4))*a)
    return s.stack(out)


def positive(x:I)->I:
    return I(np.maximum(0,x.lo),np.maximum(0,x.hi))


def positive_moment(x:I,r:F,power:int)->I:
    if r==0:
        y=positive(x);return y if power==1 else y.square()
    # Exact antiderivative under a uniform innovation. Outward operations
    # enclose cancellation rather than silently treating it as exact.
    a,b=positive(x+r),positive(x-r)
    if power==1:return (a.square()-b.square())/s.c.rat_i(4*r)
    if power==2:return (a.square()*a-b.square()*b)/s.c.rat_i(6*r)
    raise ValueError('Only powers one and two are implemented')


def terminal_expectation(x:I,a:I)->I:
    d=x.lo.shape[-1];y=deterministic_next(x,a)
    signs=[1 if j%2==0 else -1 for j in range(d)]
    yroll=I(np.roll(y.lo,-1,axis=-1),np.roll(y.hi,-1,axis=-1))
    var=F(1,3072)
    const=4*var+var*sum((signs[j]-signs[(j+1)%d])**2 for j in range(d))/(4*d)
    b=F(1,2)-s.isum(y)*s.c.rat_i(F(2,d))
    r=abs(F(sum(signs),16*d))
    return s.isum((y-F(5,8)).square())*s.c.rat_i(F(4,d))+s.isum((y-yroll).square())*s.c.rat_i(F(1,4*d))+s.c.rat_i(const)+2*positive_moment(b,r,2)


def final_q(x:I,a:I,p:int)->I:
    return s.costs(x,a,p)+float(BETA)*terminal_expectation(x,a)


def final_derivative(x:I,a:I,p:int)->I:
    d=x.lo.shape[-1];y=deterministic_next(x,a)
    gamma=[F(1,2) if j%2==0 else F(1,4) for j in range(d)]
    val=I.point(np.zeros_like(a.lo))
    for j in range(d):
        val=val+s.c.rat_i(F(8,d)*gamma[j])*(s.col(y,j)-F(5,8))
        val=val+s.c.rat_i((gamma[j]-gamma[(j+1)%d])/F(2*d))*(s.col(y,j)-s.col(y,(j+1)%d))
    b=F(1,2)-s.isum(y)*s.c.rat_i(F(2,d))
    r=abs(F(sum(1 if j%2==0 else -1 for j in range(d)),16*d))
    val=val-s.c.rat_i(F(8,d)*sum(gamma))*positive_moment(b,r,1)
    return 2*p*a+16*a.square()*a+float(BETA)*val


def final_difference(x:I,a:I,b:I,p:int)->I:
    """Centered actual Q(a)-Q(b); state-only costs cancel analytically."""
    d=x.lo.shape[-1];zero=I.point(np.zeros_like(a.lo));u=deterministic_next(x,zero)
    gam=[F(1,2) if j%2==0 else F(1,4) for j in range(d)]
    delta=a-b;plus=a+b
    out=delta*plus*(p+4*(a.square()+b.square()))
    t=zero
    for j in range(d):
        t=t+s.c.rat_i(F(4,d)*gam[j])*delta*(2*s.col(u,j)+s.c.rat_i(gam[j])*plus-F(5,4))
        dg=gam[j]-gam[(j+1)%d]
        t=t+s.c.rat_i(dg/F(4*d))*delta*(2*(s.col(u,j)-s.col(u,(j+1)%d))+s.c.rat_i(dg)*plus)
    ba=F(1,2)-s.isum(deterministic_next(x,a))*s.c.rat_i(F(2,d))
    bb=F(1,2)-s.isum(deterministic_next(x,b))*s.c.rat_i(F(2,d))
    r=abs(F(sum(1 if j%2==0 else -1 for j in range(d)),16*d))
    if r==0:
        pa,pb=positive(ba),positive(bb)
        t=t+2*(pa-pb)*(pa+pb)
    else:t=t+2*(positive_moment(ba,r,2)-positive_moment(bb,r,2))
    return out+float(BETA)*t


def exact_q(x,a,p,derivative=False):
    """Rational reference and rare exact point-comparison fallback."""
    x=list(map(F,x));a=F(a);d=len(x);gam=[F(1,2) if j%2==0 else F(1,4) for j in range(d)]
    y=[F(1,16)+x[j]/2+x[(j+1)%d]/8+x[j]*(1-x[(j+1)%d])/16+gam[j]*a for j in range(d)]
    signs=[1 if j%2==0 else -1 for j in range(d)]
    b=F(1,2)-2*sum(y)/d;r=abs(F(sum(signs),16*d));pos=lambda z:max(F(0),z)
    if derivative:
        ep=pos(b) if r==0 else (pos(b+r)**2-pos(b-r)**2)/(4*r)
        return 2*p*a+16*a**3+BETA*(8*sum(gam[j]*(y[j]-F(5,8)) for j in range(d))/d+sum((gam[j]-gam[(j+1)%d])*(y[j]-y[(j+1)%d]) for j in range(d))/(2*d)-8*sum(gam)*ep/d)
    ep=pos(b)**2 if r==0 else (pos(b+r)**3-pos(b-r)**3)/(6*r)
    var=F(1,3072);g=4*sum((z-F(5,8))**2 for z in y)/d+sum((y[j]-y[(j+1)%d])**2 for j in range(d))/(4*d)+2*ep+4*var+var*sum((signs[j]-signs[(j+1)%d])**2 for j in range(d))/(4*d)
    c=2*sum((z-F(5,8))**2 for z in x)/d+sum((x[j]-x[(j+1)%d])**2 for j in range(d))/(4*d)+2*pos(F(1,2)-2*sum(x)/d)**2+p*a*a+4*a**4
    return c+BETA*g


def proposal(bins:np.ndarray,bits:int,p:int,counts:dict)->np.ndarray:
    n,d=bins.shape;nbit=2**bits;center=(bins+.5)/nbit;x=I.point(center)
    capidx=(QUANTUM*(nbit*d+np.sum(bins,axis=1)))//(8*d*nbit)
    lo=np.zeros(n,dtype=np.int64);hi=capidx.astype(np.int64)
    # Find the first nonnegative derivative index, including boundary optima.
    for _ in range(12):
        active=lo<hi
        if not np.any(active):break
        ids=np.flatnonzero(active);mid=(lo[ids]+hi[ids])//2
        val=final_derivative(I.point(center[ids]),I.point(mid/QUANTUM),p)
        neg=val.hi<0;pos=val.lo>=0;amb=~(neg|pos)
        counts['derivative_queries']=counts.get('derivative_queries',0)+len(ids)
        for k in np.flatnonzero(amb):
            neg[k]=exact_q(center[ids[k]],F(int(mid[k]),QUANTUM),p,True)<0
            counts['exact_point_fallbacks']=counts.get('exact_point_fallbacks',0)+1
        lo[ids[neg]]=mid[neg]+1;hi[ids[~neg]]=mid[~neg]
    a=lo/QUANTUM;b=np.maximum(0,lo-1)/QUANTUM
    diff=final_difference(x,I.point(a),I.point(b),p)
    take=diff.hi<0;amb=(diff.lo<=0)&(diff.hi>=0)&(a!=b)
    for k in np.flatnonzero(amb):
        take[k]=exact_q(center[k],F(float(a[k])),p)<exact_q(center[k],F(float(b[k])),p)
        counts['exact_point_fallbacks']=counts.get('exact_point_fallbacks',0)+1
    counts['proposal_cost_comparisons']=counts.get('proposal_cost_comparisons',0)+n
    return np.where(take,a,b)


class AcquiredPolicy:
    """Original acquired actor with an optional cell-wise terminal safety gate."""
    def __init__(self,payload,bits=8):
        self.raw=payload;self.T=payload['T'];self.p=payload['price'];self.d=payload['dimension'];self.bits=bits
        self.models=[s.Model.load(m) for m in payload['models'][:-1]];self.actors=payload['actors'];self.counts={}
    def _cell_base(self,t,bins):
        nbit=2**self.bits;obs=I.point((bins+.5)/nbit)
        raw,amb=self.models[t].actor(obs,self.actors[t]);assert not amb
        capidx=(QUANTUM*(nbit*self.d+np.sum(bins,axis=1)))//(8*self.d*nbit)
        action=np.minimum(np.maximum(0,np.floor(raw.lo*QUANTUM)/QUANTUM),capidx/QUANTUM)
        self.counts['actor_queries']=self.counts.get('actor_queries',0)+len(bins)
        self.counts['capacity_or_quantum_repairs']=self.counts.get('capacity_or_quantum_repairs',0)+int(np.count_nonzero(action<raw.lo))
        return action
    def cell_actions(self,t,bins,repair=False):
        base=self._cell_base(t,bins)
        if not repair:return base,base
        assert t==self.T-1
        proposed=proposal(bins,self.bits,self.p,self.counts)
        cell=I(bins/2**self.bits,(bins+1)/2**self.bits)
        diff=final_difference(cell,I.point(proposed),I.point(base),self.p)
        accepted=(diff.hi<=0)&(base!=proposed)
        out=np.where(accepted,proposed,base)
        self.counts['gate_queries']=self.counts.get('gate_queries',0)+len(bins)
        self.counts['strict_action_changes']=self.counts.get('strict_action_changes',0)+int(np.count_nonzero(accepted))
        self.counts['blocked_action_changes']=self.counts.get('blocked_action_changes',0)+int(np.count_nonzero((base!=proposed)&~accepted))
        self.counts['candidate_equal_incumbent']=self.counts.get('candidate_equal_incumbent',0)+int(np.count_nonzero(base==proposed))
        if np.any(accepted):
            hi=float(np.max(diff.hi[accepted]));self.counts['largest_accepted_advantage_upper']=max(self.counts.get('largest_accepted_advantage_upper',-math.inf),hi)
        return base,out
    def action(self,t,x,repair=False):
        n=2**self.bits;first=np.clip(np.floor(x.lo*n).astype(np.int64),0,n-1);last=np.clip(np.floor(x.hi*n).astype(np.int64),0,n-1)
        same=np.all(first==last,axis=1);a=np.empty(len(first));b=np.empty(len(first));ar=np.empty(len(first));br=np.empty(len(first))
        ids=np.flatnonzero(same)
        if len(ids):
            aa,rr=self.cell_actions(t,first[ids],repair);a[ids]=b[ids]=aa;ar[ids]=br[ids]=rr
        for k in np.flatnonzero(~same):
            num=math.prod(int(v-u+1) for u,v in zip(first[k],last[k]));self.counts['ambiguous_acquisitions']=self.counts.get('ambiguous_acquisitions',0)+1
            if num>4096:
                a[k]=ar[k]=0;b[k]=br[k]=s.cap(I(x.lo[k:k+1],x.hi[k:k+1])).hi[0]
                self.counts['conservative_hull_fallbacks']=self.counts.get('conservative_hull_fallbacks',0)+1
            else:
                bins=np.array(list(itertools.product(*[range(int(u),int(v)+1) for u,v in zip(first[k],last[k])])),dtype=np.int64)
                aa,rr=self.cell_actions(t,bins,repair);a[k]=aa.min();b[k]=aa.max();ar[k]=rr.min();br[k]=rr.max()
        return I(a,b),I(ar,br)
    def paired_scores(self,x,z):
        total=I.point(np.zeros(len(x.lo)));stages=[]
        for t in range(self.T-1):
            a,_=self.action(t,x);stages.append(float(np.mean(a.midpoint())))
            total=total+float(BETA**t)*s.costs(x,a,self.p);x=s.transition(x,a,z[t])
        a,r=self.action(self.T-1,x,True)
        old=total+float(BETA**(self.T-1))*final_q(x,a,self.p)
        new=total+float(BETA**(self.T-1))*final_q(x,r,self.p)
        delta=float(BETA**(self.T-1))*final_difference(x,r,a,self.p)
        # The cell gate proves nonpositive TRUE advantage. Intersection uses
        # that independently verified support, not an empirical clipping rule.
        delta=I(delta.lo,np.minimum(0,delta.hi))
        if np.any(delta.lo>delta.hi):raise AssertionError('Invalid advantage enclosure')
        return old,new,delta,{'last_old_action_mean':float(np.mean(a.midpoint())),'last_new_action_mean':float(np.mean(r.midpoint())),'last_state_mean':np.mean(x.midpoint(),axis=0).tolist(),'earlier_action_means':stages}


def plan(d,T,p,epsilon,cap=(64,16,16)):
    a,b,q=s.coefficients(d,T,p);reserve=F(1,10**8);E=F(epsilon)-reserve
    axes=[[2**k for k in range(7) if 2**k<=v] for v in cap];axes[0]=[v for v in axes[0] if v>=4]
    candidates=[]
    for N,K,M in itertools.product(*axes):
        bound=a/N+b/K+q/M
        if bound<=E:candidates.append((T*(N+1)**d*(K+1)*M,N,K,M,bound))
    if not candidates:return {'status':'cap-failure','epsilon':str(epsilon),'cap':cap,'coefficient':list(map(str,(a,b,q)))}
    w,N,K,M,bound=min(candidates)
    return {'status':'planned','N':N,'K':K,'M':M,'target':str(epsilon),'proxy_queries':w,'primitive_bound':str(bound),'arithmetic_reserve':str(reserve),'coefficients':list(map(str,(a,b,q))),'finite_feasible_allocations':len(candidates),'planning_candidates':math.prod(map(len,axes)),'objective':'exact finite Bellman state-action-innovation proxy; not measured total work'}


def surplus_rung(N,K,M,T,p):
    """Sequential own-future midpoint-surplus knot insertion, genuinely adaptive."""
    d=2;counts=dict(bellman_queries=0,innovation_midpoints=0,stage_cost_evaluations=0,terminal_cost_evaluations=0,pilot_bellman_queries=0,nearest_cell_vertices=0,surplus_comparisons=0,inserted_knots=0)
    C,G,Fx,Fa,Ca,Ka=s.moduli(d,p);geometry=[]
    def grid(future=None):
        axes=[list(np.arange(5)/4),list(np.arange(5)/4)];cache={};maxerr=F(0);history=[]
        def values(nodes):
            nonlocal maxerr
            missing=list(dict.fromkeys(tuple(map(float,x)) for x in nodes if tuple(map(float,x)) not in cache))
            if missing:
                pts=np.array(missing)
                if future is None:
                    vals,e=s.c.rounded_labels(s.costs(I.point(pts),I.point(np.zeros(len(pts))),p,True));acts=np.zeros(len(pts));counts['terminal_cost_evaluations']+=len(pts)
                else:vals,acts,e=s.queries(pts,K,M,p,future,counts)
                maxerr=max(maxerr,e)
                for key,v,act in zip(missing,vals,acts):cache[key]=(float(v),float(act))
            return np.array([cache[tuple(map(float,x))][0] for x in nodes])
        while any(len(ax)<N+1 for ax in axes):
            picks=[]
            for j in range(d):
                if len(axes[j])>=N+1:continue
                other=axes[1-j]
                for k,(a,b) in enumerate(zip(axes[j],axes[j][1:])):
                    pts=[]
                    for v in (a,(a+b)/2,b):
                        pts.extend([(v,z) if j==0 else (z,v) for z in other])
                    y=values(np.array(pts)).reshape(3,len(other));err=float(np.max(np.abs(y[1]-(y[0]+y[2])/2)))
                    counts['surplus_comparisons']+=len(other);picks.append((-err,j,k,a,b))
            neg,j,k,a,b=min(picks);axes[j].insert(k+1,(a+b)/2);counts['inserted_knots']+=1
            history.append({'axis':j,'interval':[a,b],'absolute_midpoint_surplus':-neg})
        nodes=np.array(list(itertools.product(*axes)));y=values(nodes);acts=np.array([cache[tuple(x)][1] for x in nodes]);counts['pilot_bellman_queries']+=max(0,len(cache)-len(nodes))*(K+1) if future is not None else 0
        return [np.array(a) for a in axes],y,acts,maxerr,history
    axes,y,_,eT,hist=grid();future=s.Model(axes,y,'fvi',G);models=[None]*T+[future];actors=[None]*T;rows=[None]*T;terminal_width=2*eT+2*G*future.h;k=F(1,8*K)+F(1,2**48);geometry.append({'date':T,'insertions':hist})
    for t in range(T-1,-1,-1):
        A=C+BETA*Fx*future.L;D=Ca+BETA*Fa*future.L;L=A+D*Ka
        axes,y,act,e,hist=grid(future);current=s.Model(axes,y,'fvi',L);extra,nn=s.old.nearest_excess(current,L);counts['nearest_cell_vertices']+=nn
        u=D*k+e+L*current.h;selected=e+extra
        rows[t]={'date':t,'A':str(A),'D':str(D),'L_graph':str(L),'query_error':str(e),'state_cover':str(current.h),'action_cover':str(k),'optimal_residual_upper':str(u),'selected_policy_upper':str(selected),'nearest_excess':str(extra),'component':str(u+selected)}
        models[t]=current;actors[t]=act.tolist();geometry.append({'date':t,'insertions':hist});future=current
    bound=sum((BETA**t*F(row['component']) for t,row in enumerate(rows)),F(0))+BETA**T*terminal_width
    deployment=s.old.deployment_check(models[:-1],actors)
    for name in models[0].counts:counts[name]=sum(m.counts[name] for m in models)
    counts.update(stored_critic_labels=sum(m.S for m in models),stored_actor_scalars=sum(map(len,actors)),maximum_coefficient_bits=max(m.bits for m in models))
    nonuniform=sum(any(len(set(np.diff(a)))>1 for a in m.axes) for m in models)
    return {'method':'surplus-fvi','N':N,'K':K,'M':M,'T':T,'price':p,'dimension':d,'models':[m.payload() for m in models],'actors':actors,'rows':rows,'terminal_error':str(eT),'terminal_width':str(terminal_width),'policy_bound_exact':str(bound),'policy_bound_upper':s.c.upper(bound),'nonuniform_date_models':nonuniform,'deployment':deployment,'counts':counts,'adaptation_history':geometry}
